\medskip\noindent\textbf{Q.3  Values of the unknown for which the rank is 3 / less than 3}\par\smallskip
A $3\times3$ matrix has rank $3$ iff $\det\neq 0$.


\noindent\textbf{(1)} $\det A=- \left(\lambda - 3\right)^{2} \left(\lambda - 2\right)$.
\begin{itemize}\item (i) rank $=3$ for $\lambda\neq 2, 3$.
\item (ii) rank $<3$ for $\lambda=2$ (then rank $=2$); $\lambda=3$ (then rank $=2$).\end{itemize}

\noindent\textbf{(2)} $\det A=13 \lambda - 55$.
\begin{itemize}\item (i) rank $=3$ for $\lambda\neq \frac{55}{13}$.
\item (ii) rank $<3$ for $\lambda=\frac{55}{13}$ (then rank $=2$).\end{itemize}

\noindent\textbf{(3)} $\det A=\lambda^{3} - 77 \lambda + 180$.
The cubic has no rational root; its real roots are $\lambda\approx -9.769, 2.554, 7.215$.\begin{itemize}\item (i) rank $=3$ for every $\lambda$ except these three values.\item (ii) rank $<3$ exactly for these three values; there rank $=2$ (e.g. the minor $\lambda^2-8\ne0$ there).\end{itemize}

\noindent\textbf{(4)} $A=\begin{pmatrix}1&2&3\\2&4&a\\3&6&b\end{pmatrix}$. $R_2\to R_2-2R_1,\ R_3\to R_3-3R_1$ gives $\begin{pmatrix}1&2&3\\0&0&a-6\\0&0&b-9\end{pmatrix}$.
\begin{itemize}\item (i) rank $=3$: \textbf{never} (columns 1 and 2 are always proportional, $\det A=0$ for all $a,b$).
\item (ii) rank $<3$: for all $a,b$. In particular rank $=1$ if $a=6$ and $b=9$, and rank $=2$ otherwise.\end{itemize}
