\subsection*{Solutions --- Tutorial No. 11}
\medskip\noindent\textbf{Q.1  Orthogonal diagonalization of symmetric matrices}\par\smallskip
For real symmetric $A$: eigenvectors of distinct eigenvalues are orthogonal; inside an eigenspace use Gram--Schmidt; normalise. The modal matrix $P$ (orthonormal eigenvectors as columns) is orthogonal and $D=P^TAP=P^{-1}AP$.

\noindent\textbf{a)} $A=\begin{pmatrix}2 & 1 \\ 1 & 2\end{pmatrix}$, $\det(A-\lambda I)=\left(\lambda - 3\right) \left(\lambda - 1\right)$, eigenvalues $1$ (mult. 1), $3$ (mult. 1).
\begin{itemize}\item Orthonormal eigenvectors: $\lambda=1$: $\begin{pmatrix}- \frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2}\end{pmatrix}$; $\lambda=3$: $\begin{pmatrix}\frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2}\end{pmatrix}$.
\item Modal matrix $P=\begin{pmatrix}- \frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2}\end{pmatrix}$, and $P^TAP=D=\begin{pmatrix}1 & 0 \\ 0 & 3\end{pmatrix}$.\end{itemize}

\noindent\textbf{b)} $A=\begin{pmatrix}-3 & 0 \\ 0 & 2\end{pmatrix}$, $\det(A-\lambda I)=\left(\lambda - 2\right) \left(\lambda + 3\right)$, eigenvalues $-3$ (mult. 1), $2$ (mult. 1).
\begin{itemize}\item Orthonormal eigenvectors: $\lambda=-3$: $\begin{pmatrix}1 \\ 0\end{pmatrix}$; $\lambda=2$: $\begin{pmatrix}0 \\ 1\end{pmatrix}$.
\item Modal matrix $P=\begin{pmatrix}1 & 0 \\ 0 & 1\end{pmatrix}$, and $P^TAP=D=\begin{pmatrix}-3 & 0 \\ 0 & 2\end{pmatrix}$.\end{itemize}

\noindent\textbf{c)} $A=\begin{pmatrix}3 & 2 & 4 \\ 2 & 0 & 2 \\ 4 & 2 & 3\end{pmatrix}$, $\det(A-\lambda I)=\left(\lambda - 8\right) \left(\lambda + 1\right)^{2}$, eigenvalues $-1$ (mult. 2), $8$ (mult. 1).
\begin{itemize}\item Orthonormal eigenvectors: $\lambda=-1$: $\begin{pmatrix}- \frac{\sqrt{5}}{5} \\ \frac{2 \sqrt{5}}{5} \\ 0\end{pmatrix}$, $\begin{pmatrix}- \frac{4 \sqrt{5}}{15} \\ - \frac{2 \sqrt{5}}{15} \\ \frac{\sqrt{5}}{3}\end{pmatrix}$; $\lambda=8$: $\begin{pmatrix}\frac{2}{3} \\ \frac{1}{3} \\ \frac{2}{3}\end{pmatrix}$.
\item Modal matrix $P=\begin{pmatrix}- \frac{\sqrt{5}}{5} & - \frac{4 \sqrt{5}}{15} & \frac{2}{3} \\ \frac{2 \sqrt{5}}{5} & - \frac{2 \sqrt{5}}{15} & \frac{1}{3} \\ 0 & \frac{\sqrt{5}}{3} & \frac{2}{3}\end{pmatrix}$, and $P^TAP=D=\begin{pmatrix}-1 & 0 & 0 \\ 0 & -1 & 0 \\ 0 & 0 & 8\end{pmatrix}$.\end{itemize}

\noindent\textbf{d)} $A=\begin{pmatrix}-2 & 5 & 4 \\ 5 & 7 & 5 \\ 4 & 5 & -2\end{pmatrix}$, $\det(A-\lambda I)=\left(\lambda - 12\right) \left(\lambda + 3\right) \left(\lambda + 6\right)$, eigenvalues $-6$ (mult. 1), $-3$ (mult. 1), $12$ (mult. 1).
\begin{itemize}\item Orthonormal eigenvectors: $\lambda=-6$: $\begin{pmatrix}- \frac{\sqrt{2}}{2} \\ 0 \\ \frac{\sqrt{2}}{2}\end{pmatrix}$; $\lambda=-3$: $\begin{pmatrix}\frac{\sqrt{3}}{3} \\ - \frac{\sqrt{3}}{3} \\ \frac{\sqrt{3}}{3}\end{pmatrix}$; $\lambda=12$: $\begin{pmatrix}\frac{\sqrt{6}}{6} \\ \frac{\sqrt{6}}{3} \\ \frac{\sqrt{6}}{6}\end{pmatrix}$.
\item Modal matrix $P=\begin{pmatrix}- \frac{\sqrt{2}}{2} & \frac{\sqrt{3}}{3} & \frac{\sqrt{6}}{6} \\ 0 & - \frac{\sqrt{3}}{3} & \frac{\sqrt{6}}{3} \\ \frac{\sqrt{2}}{2} & \frac{\sqrt{3}}{3} & \frac{\sqrt{6}}{6}\end{pmatrix}$, and $P^TAP=D=\begin{pmatrix}-6 & 0 & 0 \\ 0 & -3 & 0 \\ 0 & 0 & 12\end{pmatrix}$.\end{itemize}

\noindent\textbf{e)} $A=\begin{pmatrix}6 & -2 & 2 \\ -2 & 3 & -1 \\ 2 & -1 & 3\end{pmatrix}$, $\det(A-\lambda I)=\left(\lambda - 8\right) \left(\lambda - 2\right)^{2}$, eigenvalues $2$ (mult. 2), $8$ (mult. 1).
\begin{itemize}\item Orthonormal eigenvectors: $\lambda=2$: $\begin{pmatrix}\frac{\sqrt{5}}{5} \\ \frac{2 \sqrt{5}}{5} \\ 0\end{pmatrix}$, $\begin{pmatrix}- \frac{\sqrt{30}}{15} \\ \frac{\sqrt{30}}{30} \\ \frac{\sqrt{30}}{6}\end{pmatrix}$; $\lambda=8$: $\begin{pmatrix}\frac{\sqrt{6}}{3} \\ - \frac{\sqrt{6}}{6} \\ \frac{\sqrt{6}}{6}\end{pmatrix}$.
\item Modal matrix $P=\begin{pmatrix}\frac{\sqrt{5}}{5} & - \frac{\sqrt{30}}{15} & \frac{\sqrt{6}}{3} \\ \frac{2 \sqrt{5}}{5} & \frac{\sqrt{30}}{30} & - \frac{\sqrt{6}}{6} \\ 0 & \frac{\sqrt{30}}{6} & \frac{\sqrt{6}}{6}\end{pmatrix}$, and $P^TAP=D=\begin{pmatrix}2 & 0 & 0 \\ 0 & 2 & 0 \\ 0 & 0 & 8\end{pmatrix}$.\end{itemize}

\noindent\textbf{f)} $A=\begin{pmatrix}3 & -1 & 1 \\ -1 & 5 & -1 \\ 1 & -1 & 3\end{pmatrix}$, $\det(A-\lambda I)=\left(\lambda - 6\right) \left(\lambda - 3\right) \left(\lambda - 2\right)$, eigenvalues $2$ (mult. 1), $3$ (mult. 1), $6$ (mult. 1).
\begin{itemize}\item Orthonormal eigenvectors: $\lambda=2$: $\begin{pmatrix}- \frac{\sqrt{2}}{2} \\ 0 \\ \frac{\sqrt{2}}{2}\end{pmatrix}$; $\lambda=3$: $\begin{pmatrix}\frac{\sqrt{3}}{3} \\ \frac{\sqrt{3}}{3} \\ \frac{\sqrt{3}}{3}\end{pmatrix}$; $\lambda=6$: $\begin{pmatrix}\frac{\sqrt{6}}{6} \\ - \frac{\sqrt{6}}{3} \\ \frac{\sqrt{6}}{6}\end{pmatrix}$.
\item Modal matrix $P=\begin{pmatrix}- \frac{\sqrt{2}}{2} & \frac{\sqrt{3}}{3} & \frac{\sqrt{6}}{6} \\ 0 & \frac{\sqrt{3}}{3} & - \frac{\sqrt{6}}{3} \\ \frac{\sqrt{2}}{2} & \frac{\sqrt{3}}{3} & \frac{\sqrt{6}}{6}\end{pmatrix}$, and $P^TAP=D=\begin{pmatrix}2 & 0 & 0 \\ 0 & 3 & 0 \\ 0 & 0 & 6\end{pmatrix}$.\end{itemize}
\medskip\noindent\textbf{Q.2  Canonical forms of quadratic forms}\par\smallskip
Write $Q=\mathbf x^TA\mathbf x$ with $A$ symmetric; with $\mathbf x=P\mathbf y$ ($P$ orthogonal) $Q=\lambda_1y_1^2+\lambda_2y_2^2+\dots$. Rank $r=$ number of non-zero $\lambda_i$; index $p=$ number of positive $\lambda_i$; signature $s=p-(r-p)=2p-r$. The surface type is stated for $Q=1$.

\noindent\textbf{1.} $Q=6 x^{2} - 4 x y + 4 x z + 3 y^{2} - 2 y z + 3 z^{2}$, $A=\begin{pmatrix}6 & -2 & 2 \\ -2 & 3 & -1 \\ 2 & -1 & 3\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $2$, $2$, $8$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}\frac{\sqrt{5}}{5} & - \frac{\sqrt{30}}{15} & \frac{\sqrt{6}}{3} \\ \frac{2 \sqrt{5}}{5} & \frac{\sqrt{30}}{30} & - \frac{\sqrt{6}}{6} \\ 0 & \frac{\sqrt{30}}{6} & \frac{\sqrt{6}}{6}\end{pmatrix}$.
\item Canonical form: $Q=2 y_{1}^{2} + 2 y_{2}^{2} + 8 y_{3}^{2}$.
\item Rank $=3$, index $=3$, signature $=3$; nature: \textbf{positive definite}. Type ($Q=1$): ellipsoid.\end{itemize}

\noindent\textbf{2.} $Q=2 x^{2} - 2 x y + 2 x z + 2 y^{2} - 2 y z + 2 z^{2}$, $A=\begin{pmatrix}2 & -1 & 1 \\ -1 & 2 & -1 \\ 1 & -1 & 2\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $1$, $1$, $4$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}\frac{\sqrt{2}}{2} & - \frac{\sqrt{6}}{6} & \frac{\sqrt{3}}{3} \\ \frac{\sqrt{2}}{2} & \frac{\sqrt{6}}{6} & - \frac{\sqrt{3}}{3} \\ 0 & \frac{\sqrt{6}}{3} & \frac{\sqrt{3}}{3}\end{pmatrix}$.
\item Canonical form: $Q=y_{1}^{2} + y_{2}^{2} + 4 y_{3}^{2}$.
\item Rank $=3$, index $=3$, signature $=3$; nature: \textbf{positive definite}. Type ($Q=1$): ellipsoid.\end{itemize}

\noindent\textbf{3.} $Q=2 x^{2} + 2 x y + 2 x z + 2 y^{2} - 2 y z + 2 z^{2}$, $A=\begin{pmatrix}2 & 1 & 1 \\ 1 & 2 & -1 \\ 1 & -1 & 2\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $0$, $3$, $3$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}- \frac{\sqrt{3}}{3} & \frac{\sqrt{2}}{2} & \frac{\sqrt{6}}{6} \\ \frac{\sqrt{3}}{3} & \frac{\sqrt{2}}{2} & - \frac{\sqrt{6}}{6} \\ \frac{\sqrt{3}}{3} & 0 & \frac{\sqrt{6}}{3}\end{pmatrix}$.
\item Canonical form: $Q=3 y_{2}^{2} + 3 y_{3}^{2}$.
\item Rank $=2$, index $=2$, signature $=2$; nature: \textbf{positive semi-definite}. Type ($Q=1$): elliptic cylinder.\end{itemize}

\noindent\textbf{4.} $Q=5 x^{2} + 2 x z - 2 y^{2} + 5 z^{2}$, $A=\begin{pmatrix}5 & 0 & 1 \\ 0 & -2 & 0 \\ 1 & 0 & 5\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $-2$, $4$, $6$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}0 & - \frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2} \\ 1 & 0 & 0 \\ 0 & \frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2}\end{pmatrix}$.
\item Canonical form: $Q=- 2 y_{1}^{2} + 4 y_{2}^{2} + 6 y_{3}^{2}$.
\item Rank $=3$, index $=2$, signature $=1$; nature: \textbf{indefinite}. Type ($Q=1$): hyperboloid of one sheet.\end{itemize}

\noindent\textbf{5.} $Q=2 x^{2} + 4 x y + 2 y^{2} + z^{2}$, $A=\begin{pmatrix}2 & 2 & 0 \\ 2 & 2 & 0 \\ 0 & 0 & 1\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $0$, $1$, $4$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}- \frac{\sqrt{2}}{2} & 0 & \frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2} & 0 & \frac{\sqrt{2}}{2} \\ 0 & 1 & 0\end{pmatrix}$.
\item Canonical form: $Q=y_{2}^{2} + 4 y_{3}^{2}$.
\item Rank $=2$, index $=2$, signature $=2$; nature: \textbf{positive semi-definite}. Type ($Q=1$): elliptic cylinder.\end{itemize}

\noindent\textbf{6.} $Q=2 x y + 2 x z + 2 y z$, $A=\begin{pmatrix}0 & 1 & 1 \\ 1 & 0 & 1 \\ 1 & 1 & 0\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $-1$, $-1$, $2$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}- \frac{\sqrt{2}}{2} & - \frac{\sqrt{6}}{6} & \frac{\sqrt{3}}{3} \\ \frac{\sqrt{2}}{2} & - \frac{\sqrt{6}}{6} & \frac{\sqrt{3}}{3} \\ 0 & \frac{\sqrt{6}}{3} & \frac{\sqrt{3}}{3}\end{pmatrix}$.
\item Canonical form: $Q=- y_{1}^{2} - y_{2}^{2} + 2 y_{3}^{2}$.
\item Rank $=3$, index $=1$, signature $=-1$; nature: \textbf{indefinite}. Type ($Q=1$): hyperboloid of two sheets.\end{itemize}

\noindent\textbf{7.} $Q=- 2 x^{2} - 4 x y - 2 y^{2} - z^{2}$, $A=\begin{pmatrix}-2 & -2 & 0 \\ -2 & -2 & 0 \\ 0 & 0 & -1\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $-4$, $-1$, $0$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}\frac{\sqrt{2}}{2} & 0 & - \frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2} & 0 & \frac{\sqrt{2}}{2} \\ 0 & 1 & 0\end{pmatrix}$.
\item Canonical form: $Q=- 4 y_{1}^{2} - y_{2}^{2}$.
\item Rank $=2$, index $=0$, signature $=-2$; nature: \textbf{negative semi-definite}. Type ($Q=1$): no real points; $Q=-1$ is an elliptic cylinder.\end{itemize}

\noindent\textbf{8.} $Q=2 x^{2} - 2 x z + 2 y^{2} + 2 z^{2}$, $A=\begin{pmatrix}2 & 0 & -1 \\ 0 & 2 & 0 \\ -1 & 0 & 2\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $1$, $2$, $3$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}\frac{\sqrt{2}}{2} & 0 & - \frac{\sqrt{2}}{2} \\ 0 & 1 & 0 \\ \frac{\sqrt{2}}{2} & 0 & \frac{\sqrt{2}}{2}\end{pmatrix}$.
\item Canonical form: $Q=y_{1}^{2} + 2 y_{2}^{2} + 3 y_{3}^{2}$.
\item Rank $=3$, index $=3$, signature $=3$; nature: \textbf{positive definite}. Type ($Q=1$): ellipsoid.\end{itemize}

\noindent\textbf{9.} $Q=- 2 x^{2} + 2 x z - 2 y^{2} - 2 z^{2}$, $A=\begin{pmatrix}-2 & 0 & 1 \\ 0 & -2 & 0 \\ 1 & 0 & -2\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $-3$, $-2$, $-1$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}- \frac{\sqrt{2}}{2} & 0 & \frac{\sqrt{2}}{2} \\ 0 & 1 & 0 \\ \frac{\sqrt{2}}{2} & 0 & \frac{\sqrt{2}}{2}\end{pmatrix}$.
\item Canonical form: $Q=- 3 y_{1}^{2} - 2 y_{2}^{2} - y_{3}^{2}$.
\item Rank $=3$, index $=0$, signature $=-3$; nature: \textbf{negative definite}. Type ($Q=1$): no real points (imaginary ellipsoid); $Q=-1$ is an ellipsoid.\end{itemize}

\noindent\textbf{10.} $Q=2 x^{2} - 2 x y + 2 y^{2}$, $A=\begin{pmatrix}2 & -1 \\ -1 & 2\end{pmatrix}$.
\begin{itemize}\item Eigenvalues: $1$, $3$; orthonormal eigenvectors (columns of $P$): $P=\begin{pmatrix}\frac{\sqrt{2}}{2} & - \frac{\sqrt{2}}{2} \\ \frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2}\end{pmatrix}$.
\item Canonical form: $Q=y_{1}^{2} + 3 y_{2}^{2}$.
\item Rank $=2$, index $=2$, signature $=2$; nature: \textbf{positive definite}. Type ($Q=1$): ellipse.\end{itemize}
